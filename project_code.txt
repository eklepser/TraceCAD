from PySide6.QtCore import QPointF

from pbe_editor.core.board_models import is_gnd_net, is_power_net

)

class BoardManager:
    def __init__(self, board, scene, canvas):
        self.board = board
        self.scene = scene
        self.canvas = canvas
        self.items = {}
        self.line_items = {}
        self.circle_items = {}
        self.arc_items = {}
        self.text_items = {}
        self.ratsnest_items = []
        self.show_ratsnest = True

    def add_module(self, module):
        item = BoardModuleGraphicsItem(module, self.canvas)
        self.scene.addItem(item)
        self.items[module.uid] = item
        return item

    def add_line(self, line):
        item = BoardLineGraphicsItem(line, self.canvas)
        self.scene.addItem(item)
        self.line_items[line.uid] = item
        return item

    def add_circle(self, circle):
        item = BoardCircleGraphicsItem(circle, self.canvas)
        self.scene.addItem(item)
        self.circle_items[circle.uid] = item
        return item

    def add_arc(self, arc):
        item = BoardArcGraphicsItem(arc, self.canvas)
        self.scene.addItem(item)
        self.arc_items[arc.uid] = item
        return item

    def add_text(self, text):
        item = BoardTextGraphicsItem(text, self.canvas)
        self.scene.addItem(item)
        self.text_items[text.uid] = item
        return item

    def remove_module(self, module):
        item = self.items.pop(module.uid, None)
        if item is not None and item.scene() is self.scene:
            self.scene.removeItem(item)

    def remove_line(self, line):
        item = self.line_items.pop(line.uid, None)
        if item is not None and item.scene() is self.scene:
            self.scene.removeItem(item)

    def remove_circle(self, circle):
        item = self.circle_items.pop(circle.uid, None)
        if item is not None and item.scene() is self.scene:
            self.scene.removeItem(item)

    def remove_arc(self, arc):
        item = self.arc_items.pop(arc.uid, None)
        if item is not None and item.scene() is self.scene:
            self.scene.removeItem(item)

    def remove_text(self, text):
        item = self.text_items.pop(text.uid, None)
        if item is not None and item.scene() is self.scene:
            self.scene.removeItem(item)

    def clear(self):
        all_items = (list(self.items.values()) + list(self.line_items.values())
                     + list(self.circle_items.values()) + list(self.arc_items.values())
                     + list(self.text_items.values()))
        for item in all_items:
            try:
                if item.scene() is self.scene:
                    self.scene.removeItem(item)
            except RuntimeError:
                pass
        self.items.clear()
        self.line_items.clear()
        self.circle_items.clear()
        self.arc_items.clear()
        self.text_items.clear()
        self.clear_ratsnest()

    def clear_ratsnest(self):
        for item in self.ratsnest_items:
            try:
                if item.scene() is self.scene:
                    self.scene.removeItem(item)
            except RuntimeError:
                pass
        self.ratsnest_items.clear()

    def rebuild_ratsnest(self):
        self.clear_ratsnest()
        if not self.show_ratsnest:
            return
        net_pins = self.board.collect_net_pins()
        for code, pins in net_pins.items():
            if len(pins) < 2:
                continue
            net_name = pins[0][0].pad_nets.get(pins[0][1])
            net_name = net_name.net_name if net_name else ""
            is_gnd = is_gnd_net(net_name)
            is_power = is_power_net(net_name) and not is_gnd
            points = [p[2] for p in pins]
            edges = build_mst_edges(points)
            for a_idx, b_idx in edges:
                module_a, pad_a, pos_a = pins[a_idx]
                module_b, pad_b, pos_b = pins[b_idx]
                item = RatsnestGraphicsItem(
                    QPointF(*pos_a), QPointF(*pos_b),
                    net_code=code, net_name=net_name,
                    is_power=is_power, is_gnd=is_gnd,
                    label_a=f"{module_a.reference}.{pad_a}",
                    label_b=f"{module_b.reference}.{pad_b}",
                )
                self.scene.addItem(item)
                self.ratsnest_items.append(item)

    def refresh_ratsnest_for_module(self, module):
        self.rebuild_ratsnest()

POWER_NET_NAMES = {"GND", "VCC", "VDD", "VSS", "+5V", "+3V3", "+3.3V", "-5V", "-12V", "+12V"}
GND_NET_NAMES = {"GND"}


def is_power_net(name: str) -> bool:
    if not name:
        return False
    return name.upper() in {n.upper() for n in POWER_NET_NAMES}


def is_gnd_net(name: str) -> bool:
    if not name:
        return False
    return name.upper() in {n.upper() for n in GND_NET_NAMES}


def _deepcopy_no_graphics(self, memo):
    cls = self.__class__
    result = cls.__new__(cls)
    memo[id(self)] = result
    for k, v in self.__dict__.items():
        if k == 'graphics_item':
            setattr(result, k, None)
        else:
            setattr(result, k, copy.deepcopy(v, memo))
    return result

class BoardModule:
    footprint: Footprint
    reference: str
    value: str
    x: float = 0.0
    y: float = 0.0
    angle: float = 0.0
    layer: str = "F.Cu"
    pad_nets: Dict[str, PadNet] = field(default_factory=dict)
    uid: int = 0
    graphics_item: object = None

    def __deepcopy__(self, memo):
        return _deepcopy_no_graphics(self, memo)

    def get_pad_absolute(self, pad_number: str):
        pad = None
        for p in self.footprint.pads:
            if p.number == pad_number:
                pad = p
                break
        if pad is None:
            return None
        rad = math.radians(self.angle)
        cos_a = math.cos(rad)
        sin_a = math.sin(rad)
        lx = pad.x
        ly = pad.y
        gx = self.x + lx * cos_a - ly * sin_a
        gy = self.y + lx * sin_a + ly * cos_a
        return gx, gy

    def get_footprint_copy(self) -> Footprint:
        return copy.deepcopy(self.footprint)

@dataclass
class Board:
    name: str = "Untitled"
    modules: List[BoardModule] = field(default_factory=list)
    lines: List[BoardLine] = field(default_factory=list)
    circles: List[BoardCircle] = field(default_factory=list)
    arcs: List[BoardArc] = field(default_factory=list)
    texts: List[BoardText] = field(default_factory=list)
    nets: Dict[int, Net] = field(default_factory=dict)
    filename: Optional[str] = None

    def get_net_by_name(self, name: str) -> Optional[Net]:
        for net in self.nets.values():
            if net.name == name:
                return net
        return None

    def get_or_create_net(self, name: str) -> Net:
        net = self.get_net_by_name(name)
        if net is not None:
            return net
        code = max(self.nets.keys()) + 1 if self.nets else 1
        net = Net(code=code, name=name)
        self.nets[code] = net
        return net

    def get_net_by_code(self, code: int) -> Optional[Net]:
        return self.nets.get(code)

    def collect_net_pins(self):
        result = {}
        for module in self.modules:
            for pad_number, pad_net in module.pad_nets.items():
                if pad_net.net_code <= 0:
                    continue
                if not pad_net.net_name:
                    continue
                pos = module.get_pad_absolute(pad_number)
                if pos is None:
                    continue
                result.setdefault(pad_net.net_code, []).append((module, pad_number, pos))
        for code in result:
            result[code].sort(key=lambda x: (x[0].reference, x[1]))
        return result

    def all_references(self):
        return {m.reference for m in self.modules}

    def auto_reference(self, prefix: str = "REF") -> str:
        existing = self.all_references()
        i = 1
        while True:
            name = f"{prefix}{i}"
            if name not in existing:
                return name
            i += 1

    def find_pad_at(self, scene_x: float, scene_y: float, tolerance: float):
        best = None
        best_dist = tolerance
        for module in self.modules:
            for pad in module.footprint.pads:
                pos = module.get_pad_absolute(pad.number)
                if pos is None:
                    continue
                dx = scene_x - pos[0]
                dy = scene_y - pos[1]
                dist = math.hypot(dx, dy)
                if dist <= best_dist:
                    best_dist = dist
                    best = (module, pad.number)
        return best

import re
from dataclasses import dataclass, field
from pathlib import Path
from typing import Dict, List


@dataclass
class NetlistComponent:
    timestamp: str
    footprint: str
    reference: str
    value: str
    pins: Dict[str, str] = field(default_factory=dict)


def parse_netlist(filepath: str) -> List[NetlistComponent]:
    content = Path(filepath).read_text(encoding='utf-8', errors='ignore')
    return parse_netlist_content(content)


def parse_netlist_content(content: str) -> List[NetlistComponent]:
    components: List[NetlistComponent] = []
    lines = content.splitlines()
    i = 0
    n = len(lines)
    while i < n:
        line = lines[i].strip()
        if not line.startswith('( /'):
            i += 1
            continue

        match = re.match(
            r'^\(\s+(/\S+)\s+(\S+)\s+(\S+)\s+(.*?)\s*$',
            line
        )
        if not match:
            i += 1
            continue

        timestamp = match.group(1)
        footprint = match.group(2)
        reference = match.group(3)
        value = match.group(4).strip()

        component = NetlistComponent(
            timestamp=timestamp,
            footprint=footprint,
            reference=reference,
            value=value,
        )

        i += 1
        while i < n:
            pin_line = lines[i].strip()
            if pin_line == ')':
                break
            pin_match = re.match(r'^\(\s*(\S+)\s+(.+?)\s*\)\s*$', pin_line)
            if pin_match:
                pin_number = pin_match.group(1)
                net_name = pin_match.group(2).strip().strip('"')
                component.pins[pin_number] = net_name
            i += 1

        components.append(component)
        i += 1

    return components

import math


def build_mst_edges(points):
    n = len(points)
    if n < 2:
        return []

    in_tree = [False] * n
    in_tree[0] = True
    edges = []

    best_dist = [float('inf')] * n
    best_from = [0] * n
    for j in range(1, n):
        dx = points[j][0] - points[0][0]
        dy = points[j][1] - points[0][1]
        best_dist[j] = math.hypot(dx, dy)
        best_from[j] = 0

    for _ in range(n - 1):
        u = -1
        best = float('inf')
        for j in range(n):
            if not in_tree[j] and best_dist[j] < best:
                best = best_dist[j]
                u = j
        if u == -1:
            break

        edges.append((best_from[u], u))
        in_tree[u] = True

        ux, uy = points[u][0], points[u][1]
        for v in range(n):
            if in_tree[v]:
                continue
            vx, vy = points[v][0], points[v][1]
            d = math.hypot(vx - ux, vy - uy)
            if d < best_dist[v]:
                best_dist[v] = d
                best_from[v] = u

    return edges

from PySide6.QtWidgets import QGraphicsItem, QStyle

def _layer_color(layer):
    if layer in LAYER_COLORS:
        return QColor(*LAYER_COLORS[layer])
    return QColor(*Colors.TEXT_VISIBLE)


def _snap(value, grid):
    if grid <= 0:
        return value
    return round(value / grid) * grid


class BoardLineGraphicsItem(QGraphicsItem):
    def __init__(self, line, canvas, parent=None):
        super().__init__(parent)
        self.model = line
        self.canvas = canvas
        self.model.graphics_item = self
        self.setFlag(QGraphicsItem.GraphicsItemFlag.ItemIsSelectable, True)
        self.setFlag(QGraphicsItem.GraphicsItemFlag.ItemIsMovable, True)
        self.setFlag(QGraphicsItem.GraphicsItemFlag.ItemSendsGeometryChanges, True)
        self.setZValue(2)
        self.setPos(line.x1, line.y1)

    def boundingRect(self):
        w = self.model.width + 0.5
        x2 = self.model.x2 - self.model.x1
        y2 = self.model.y2 - self.model.y1
        return QRectF(QPointF(0.0, 0.0), QPointF(x2, y2)).normalized().adjusted(-w, -w, w, w)

    def shape(self):
        path = QPainterPath()
        path.moveTo(0.0, 0.0)
        path.lineTo(self.model.x2 - self.model.x1, self.model.y2 - self.model.y1)
        stroker = QPainterPathStroker()
        stroker.setWidth(max(0.4, self.model.width * 2.0))
        return stroker.createStroke(path)

    def paint(self, painter, option, widget):
        color = QColor(*Colors.SELECTED) if (option.state & QStyle.StateFlag.State_Selected) else _layer_color(self.model.layer)
        pen = QPen(color, max(0.05, self.model.width))
        pen.setCosmetic(False)
        painter.setPen(pen)
        painter.drawLine(QLineF(0.0, 0.0,
                                self.model.x2 - self.model.x1,
                                self.model.y2 - self.model.y1))

    def itemChange(self, change, value):
        if change == QGraphicsItem.GraphicsItemChange.ItemPositionChange:
            grid = self.canvas.grid_size
            return QPointF(_snap(value.x(), grid), _snap(value.y(), grid))
        if change == QGraphicsItem.GraphicsItemChange.ItemPositionHasChanged:
            dx = value.x() - self.model.x1
            dy = value.y() - self.model.y1
            self.model.x1 = value.x()
            self.model.y1 = value.y()
            self.model.x2 += dx
            self.model.y2 += dy
            self.prepareGeometryChange()
        if change == QGraphicsItem.GraphicsItemChange.ItemSelectedHasChanged:
            self.update()
        return super().itemChange(change, value)

        self.setFlag(QGraphicsItem.GraphicsItemFlag.ItemIsMovable, True)
        self.setFlag(QGraphicsItem.GraphicsItemFlag.ItemSendsGeometryChanges, True)
        self.setZValue(2)
        self.setPos(arc.x, arc.y)

    def _local_path(self):
        pts = arc_sample_points(
            0.0, 0.0,
            self.model.sx - self.model.x,
            self.model.sy - self.model.y,
            self.model.angle, 64,
        )
        if len(pts) < 2:
            return QPainterPath()
        path = QPainterPath()
        path.moveTo(pts[0][0], pts[0][1])
        for px, py in pts[1:]:
            path.lineTo(px, py)
        return path

    def boundingRect(self):
        b = self._local_path().boundingRect()
        extra = self.model.width + 0.5
        return b.adjusted(-extra, -extra, extra, extra)

    def shape(self):
        stroker = QPainterPathStroker()
        stroker.setWidth(max(0.4, self.model.width * 2.0))
        return stroker.createStroke(self._local_path())

        pen.setCosmetic(False)
        painter.setPen(pen)
        painter.setBrush(Qt.BrushStyle.NoBrush)
        painter.drawPath(self._local_path())

    def itemChange(self, change, value):

            self.model.sy += dy
            self.prepareGeometryChange()
        if change == QGraphicsItem.GraphicsItemChange.ItemSelectedHasChanged:

        self.setZValue(2)
        self.setPos(text.x, text.y)
        self._path_cache = None
        self._path_key = None

    def _path(self):
        target_h = self.model.size * Sizes.TEXT_FONT_MULTIPLIER
        key = (self.model.text, target_h)
        if self._path_key != key:
            self._path_cache = build_text_path(self.model.text, target_h, center=True)
            self._path_key = key
        return self._path_cache

    def boundingRect(self):
        path = self._path()
        if path.isEmpty():
            return QRectF(-1, -1, 2, 2)
        b = path.boundingRect()
        return QRectF(b.left(), b.top(), b.width(), b.height()).adjusted(-1, -1, 1, 1)

    def paint(self, painter, option, widget):
        path = self._path()
        if path.isEmpty():
            return
        color = QColor(*Colors.SELECTED) if (option.state & QStyle.StateFlag.State_Selected) else _layer_color(self.model.layer)
        painter.save()
        painter.setPen(Qt.PenStyle.NoPen)
        painter.setBrush(color)
        painter.drawPath(path)
        painter.restore()

    def itemChange(self, change, value):

def _pad_block(pad, net_code, net_name):
    shape_map = {PadShape.RECT: 'R', PadShape.CIRCLE: 'C', PadShape.OVAL: 'O'}
    shape_char = shape_map.get(pad.shape, 'C')
    if pad.pad_type == PadType.STANDARD:
        drill_val = pad.drill if pad.drill else 0.8
        dr_line = f"Dr {mm_to_kicad(drill_val)} 0 0"
        at_attr = 'STD'; at_mask = '00E0FFFF'
    elif pad.pad_type == PadType.SMD:
        dr_line = "Dr 0 0 0"; at_attr = 'SMD'; at_mask = '00888000'
    else:
        dr_line = "Dr 0 0 0"; at_attr = 'CONN'; at_mask = '00808000'
    ne_line = f'Ne {net_code} "{net_name}"' if net_code > 0 and net_name else 'Ne 0 ""'
    return [
        "$PAD",
        f'Sh "{pad.number}" {shape_char} '
        f'{mm_to_kicad(pad.width)} {mm_to_kicad(pad.height)} 0 0 0',
        dr_line,
        f"At {at_attr} N {at_mask}",
        ne_line,
        f"Po {mm_to_kicad(pad.x)} {mm_to_kicad(pad.y)}",
        "$EndPAD",
    ]

def _parse_de_body(body, default_layer=28):
    m = re.search(r'^De\s+(\d+)([^\r\n]*)', body, re.MULTILINE)
    if not m:
        return layer_from_id(default_layer), 0.0
    lid = int(m.group(1))
    layer = layer_from_id(lid)
    tail = m.group(2).split()
    angle = 0.0
    if len(tail) >= 2:
        try:
            angle = float(tail[1]) / 10.0
        except ValueError:
            pass
    return layer, angle


def parse_brd_content(content: str) -> Board:
    board = Board()
    board.filename = None

    for code_str, name in re.findall(r'\$EQUIPOT\s+Na\s+(\d+)\s+"([^"]*)"', content):
        code = int(code_str)
        if code > 0:
            board.nets[code] = Net(code=code, name=name)

    line_uid = 1
    circle_uid = 1
    arc_uid = 1
    text_uid = 1

    for ds_match in re.finditer(r'\$DRAWSEGMENT(.*?)\$EndDRAWSEGMENT', content, re.DOTALL):
        body = ds_match.group(1)
        layer_name, de_angle = _parse_de_body(body)

        po_match = re.search(r'^Po\s+([-\d\s]+)', body, re.MULTILINE)
        if not po_match:
            continue
        nums = [int(x) for x in po_match.group(1).split() if x]
        if not nums:
            continue
        shape = nums[0]
        rest = nums[1:]

        if shape == 0:
            if len(rest) == 6 and rest[0] == 0:
                x1, y1, x2, y2, thick = rest[1], rest[2], rest[3], rest[4], rest[5]
            elif len(rest) >= 5:
                x1, y1, x2, y2, thick = rest[0], rest[1], rest[2], rest[3], rest[4]
            else:
                continue
            board.lines.append(BoardLine(
                x1=kicad_to_mm(x1), y1=kicad_to_mm(y1),
                x2=kicad_to_mm(x2), y2=kicad_to_mm(y2),
                layer=layer_name, width=kicad_to_mm(thick), uid=line_uid,
            ))
            line_uid += 1
            continue

        if shape in (1, 3):
            if len(rest) == 6 and rest[0] == 0:
                xc, yc, xp, yp, thick = rest[1], rest[2], rest[3], rest[4], rest[5]
            elif len(rest) >= 5:
                xc, yc, xp, yp, thick = rest[0], rest[1], rest[2], rest[3], rest[4]
            else:
                continue
            xc_m = kicad_to_mm(xc); yc_m = kicad_to_mm(yc)
            xp_m = kicad_to_mm(xp); yp_m = kicad_to_mm(yp)
            r = math.hypot(xp_m - xc_m, yp_m - yc_m)
            board.circles.append(BoardCircle(
                x=xc_m, y=yc_m, radius=r,
                layer=layer_name, width=kicad_to_mm(thick), uid=circle_uid,
            ))
            circle_uid += 1
            continue

        if shape == 2:
            if len(rest) == 7 and rest[0] == 0:
                cx, cy, sx, sy, thick = rest[1], rest[2], rest[3], rest[4], rest[6]
            elif len(rest) >= 5:

            else:
                continue
            board.arcs.append(BoardArc(
                x=kicad_to_mm(cx), y=kicad_to_mm(cy),

                angle=de_angle if de_angle != 0 else 90.0,
                layer=layer_name, width=kicad_to_mm(thick), uid=arc_uid,
            ))
            arc_uid += 1

    for tp_match in re.finditer(r'\$TEXTPCB(.*?)\$EndTEXTPCB', content, re.DOTALL):
        body = tp_match.group(1)
        layer_name, _ = _parse_de_body(body)
        te_match = re.search(r'^Te\s+"([^"]*)"', body, re.MULTILINE)
        text_value = te_match.group(1) if te_match else "Text"
        po_match = re.search(
            r'^Po\s+(-?\d+)\s+(-?\d+)\s+(\d+)\s+(\d+)\s+(\d+)\s+(\d+)',
            body, re.MULTILINE
        )
        if not po_match:
            continue
        x = kicad_to_mm(po_match.group(1))
        y = kicad_to_mm(po_match.group(2))
        w = kicad_to_mm(po_match.group(3))
        h = kicad_to_mm(po_match.group(4))
        size = max(w, h) if max(w, h) > 0 else 1.0
        board.texts.append(BoardText(
            x=x, y=y, text=text_value, size=size,
            layer=layer_name, visible=True, uid=text_uid,
        ))
        text_uid += 1

    modules = re.findall(r'\$MODULE\s+(\S+)(.*?)\$EndMODULE', content, re.DOTALL)
    next_uid = 1
    for name, body in modules:
        module = parse_module(name, body, next_uid)
        if module is None:
            continue
        next_uid += 1
        board.modules.append(module)

    return board


def parse_module(name: str, body: str, uid: int) -> Optional[BoardModule]:
    po_match = re.search(r'^Po\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)', body, re.MULTILINE)
    if not po_match:
        return None
    x = kicad_to_mm(po_match.group(1))
    y = kicad_to_mm(po_match.group(2))
    angle = float(po_match.group(3)) / 10.0

    fp = Footprint(name=name)
    reference = "REF**"
    value = "VAL**"

    t0_match = re.search(r'^T0\s+([^\r\n]+)', body, re.MULTILINE)
    t1_match = re.search(r'^T1\s+([^\r\n]+)', body, re.MULTILINE)
    if t0_match:
        ref_text = _parse_text_line(t0_match.group(1), TextType.REFERENCE)
        ref_text.locked = True
        fp.texts.append(ref_text)
        reference = ref_text.text
        fp.reference = reference
    if t1_match:
        val_text = _parse_text_line(t1_match.group(1), TextType.VALUE)
        val_text.locked = True
        fp.texts.append(val_text)
        value = val_text.text
        fp.value = value

    for ds_match in re.finditer(
        r'^DS\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)\s+(\d+)\s+(\d+)',
        body, re.MULTILINE
    ):
        x1, y1, x2, y2, thickness, layer_id = ds_match.groups()
        fp.lines.append(Line(
            x=kicad_to_mm(x1), y=kicad_to_mm(y1),
            x2=kicad_to_mm(x2), y2=kicad_to_mm(y2),
            width=kicad_to_mm(thickness),
            layer=layer_from_id(layer_id),
        ))

    for dc_match in re.finditer(
        r'^DC\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)\s+(\d+)\s+(\d+)',
        body, re.MULTILINE
    ):
        xc, yc, xp, yp, thickness, layer_id = dc_match.groups()
        xc_mm = kicad_to_mm(xc); yc_mm = kicad_to_mm(yc)
        xp_mm = kicad_to_mm(xp); yp_mm = kicad_to_mm(yp)
        radius = math.hypot(xp_mm - xc_mm, yp_mm - yc_mm)
        fp.circles.append(Circle(
            x=xc_mm, y=yc_mm, radius=radius,
            width=kicad_to_mm(thickness),
            layer=layer_from_id(layer_id),
        ))

    for da_match in re.finditer(
        r'^DA\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)\s+(\d+)\s+(\d+)',
        body, re.MULTILINE
    ):
        cx, cy, sx, sy, angle_a, thickness, layer_id = da_match.groups()
        fp.arcs.append(Arc(
            x=kicad_to_mm(cx), y=kicad_to_mm(cy),
            sx=kicad_to_mm(sx), sy=kicad_to_mm(sy),
            angle=float(angle_a) / 10.0,
            width=kicad_to_mm(thickness),
            layer=layer_from_id(layer_id),
        ))

    module = BoardModule(
        footprint=fp, reference=reference, value=value,
        x=x, y=y, angle=angle, layer="F.Cu", uid=uid,
    )

    pad_pattern = re.compile(r'\$PAD(.*?)\$EndPAD', re.DOTALL)
    for pad_match in pad_pattern.finditer(body):
        pad, pad_net = _parse_pad(pad_match.group(1))
        if pad is not None:
            fp.pads.append(pad)
            if pad_net is not None:
                module.pad_nets[pad.number] = pad_net

    return module


def _parse_text_line(text_str: str, text_type: TextType) -> Text:
    text_match = re.search(r'"([^"]*)"', text_str)
    text = text_match.group(1) if text_match else ""
    params_str = text_str[:text_match.start()].strip() if text_match else text_str
    params = params_str.split()
    x = kicad_to_mm(params[0]) if len(params) > 0 else 0
    y = kicad_to_mm(params[1]) if len(params) > 1 else 0
    w = kicad_to_mm(params[2]) if len(params) > 2 else kicad_to_mm(400)
    h = kicad_to_mm(params[3]) if len(params) > 3 else w
    size = max(w, h) if max(w, h) > 0 else 1.0
    visible = 'I' not in params[4:8]
    layer = "F.SilkS"
    for tok in params:
        try:
            layer_id = int(tok)
        except ValueError:
            continue
        if layer_id in LAYER_ID_TO_INTERNAL:
            layer = layer_from_id(layer_id)
            break
    return Text(
        x=x, y=y, text=text, size=size,
        visible=visible, text_type=text_type, locked=True,
        layer=layer,
    )


def _parse_pad(pad_body: str):
    sh_match = re.search(r'Sh\s+"([^"]*)"\s+(\w+)\s+(\d+)\s+(\d+)', pad_body)
    dr_match = re.search(r'Dr\s+(\d+)', pad_body)
    po_match = re.search(r'Po\s+(-?\d+)\s+(-?\d+)', pad_body)
    at_match = re.search(r'At\s+(\w+)', pad_body)
    ne_match = re.search(r'^Ne\s+(\d+)\s+"([^"]*)"', pad_body, re.MULTILINE)
    if not sh_match or not po_match:
        return None, None

    number = sh_match.group(1)
    shape_str = sh_match.group(2)
    size_x = kicad_to_mm(sh_match.group(3))
    size_y = kicad_to_mm(sh_match.group(4))
    x = kicad_to_mm(po_match.group(1))
    y = kicad_to_mm(po_match.group(2))
    drill = kicad_to_mm(dr_match.group(1)) if dr_match else 0

    shape_map = {'R': PadShape.RECT, 'C': PadShape.CIRCLE, 'O': PadShape.OVAL}
    shape = shape_map.get(shape_str, PadShape.CIRCLE)

    pad_type = PadType.STANDARD
    if at_match:
        attr = at_match.group(1)
        if attr == 'SMD':
            pad_type = PadType.SMD
        elif attr == 'CONN':
            pad_type = PadType.CONN

    pad = Pad(
        x=x, y=y, number=number, shape=shape, pad_type=pad_type,
        width=size_x, height=size_y,
        drill=drill if drill > 0 else None,
        layers=default_layers_for_pad_type(pad_type),
    )

    pad_net = None
    if ne_match:
        code = int(ne_match.group(1))
        name = ne_match.group(2)
        if code > 0 and name:
            pad_net = PadNet(pad_number=number, net_code=code, net_name=name)

    return pad, pad_net.

    def snap_to_grid(self, x, y):
        g = self.grid_size
        return QPointF(round(x / g) * g, round(y / g) * g)

    def set_ghost_line(self, x1, y1, x2, y2):
        self._ghost_line_start = QPointF(x1, y1)

    def drawBackground(self, painter, rect):
        super().drawBackground(painter, rect)
        if not self.show_grid:
            return
        g = self.grid_size
        scale = self.transform().m11()
        if g * scale < Sizes.MIN_CELL_PX:
            return

        left = int(rect.left() // g) * g
        right = int(rect.right() // g) * g
        top = int(rect.top() // g) * g
        bottom = int(rect.bottom() // g) * g

        step_x = g
        step_y = g
        max_points = 60
        if (right - left) / g > max_points:
            factor = 1
            while (right - left) / (g * factor) > max_points:
                factor *= 2
            step_x = g * factor
        if (bottom - top) / g > max_points:
            factor = 1
            while (bottom - top) / (g * factor) > max_points:
                factor *= 2
            step_y = g * factor

        pen = QPen(self.grid_color, 0)
        pen.setCosmetic(True)
        painter.setPen(pen)
        y = top
        while y <= bottom:
            x = left
            while x <= right:
                painter.drawPoint(QPointF(x, y))
                x += step_x
            y += step_y

    def drawForeground(self, painter, rect):
        super().drawForeground(painter, rect)
        pen = QPen(QColor(*Colors.AXIS), 0)
        pen.setCosmetic(True)
        painter.setPen(pen)
        painter.drawLine(QPointF(rect.left(), 0), QPointF(rect.right(), 0))
        painter.drawLine(QPointF(0, rect.top()), QPointF(0, rect.bottom()))

        if self._snap_pos:
            cross = Sizes.CROSS_SIZE / self.transform().m11()
            p = QPen(QColor(*Colors.SNAP_CROSS), 0)
            p.setCosmetic(True)
            painter.setPen(p)
            x = self._snap_pos.x()
            y = self._snap_pos.y()
            painter.drawLine(QPointF(x - cross, y), QPointF(x + cross, y))

    def wheelEvent(self, event: QWheelEvent):
        if event.angleDelta().y() > 0:
            if self.zoom_index < len(self.zoom_steps) - 1:
                self.zoom_index += 1
        elif self.zoom_index > 0:
            self.zoom_index -= 1
        new_zoom = self.zoom_steps[self.zoom_index]
        factor = (self.base_scale * new_zoom) / self.transform().m11()
        self.scale(factor, factor)
        self.zoom_level = new_zoom
        self.zoom_changed.emit(self.zoom_level)

    def mousePressEvent(self, event: QMouseEvent):
        if event.button() == Qt.MouseButton.MiddleButton:
            self._panning = True
            self._pan_start = event.position().toPoint()
            self.viewport().setCursor(Qt.CursorShape.ClosedHandCursor)
            event.accept()
            return
        if event.button() == Qt.MouseButton.RightButton:
            self.canvas_right_clicked.emit()
            event.accept()
            return
        if event.button() == Qt.MouseButton.LeftButton:
            # Если текущий dragMode — RubberBand (режим выделения) — пусть Qt сам обрабатывает
            if self.dragMode() == QGraphicsView.DragMode.RubberBandDrag:

                    snapped = self.snap_to_grid(scene_pos.x(), scene_pos.y())

            # Иначе — рисование: клик по пустому месту

        iy = int(self._scroll_acc_y)
        self._scroll_acc_x -= ix
        self._scroll_acc_y -= iy
        if ix or iy:
            self.horizontalScrollBar().setValue(self.horizontalScrollBar().value() + ix)
            self.verticalScrollBar().setValue(self.verticalScrollBar().value() + iy).

    def eventFilter(self, obj, event):
        if event.type() == QEvent.Type.KeyPress:
            if self._handle_key_press(event):
                return True
        elif event.type() == QEvent.Type.KeyRelease:
            if self._handle_key_release(event):
                return True
        return super().eventFilter(obj, event)

    def _handle_key_press(self, event):
        if event.key() == Qt.Key.Key_Escape:
            self._exit_to_select_mode()
            return True
        vk = event.nativeVirtualKey()
        if vk == VK_DELETE:
            self._delete_selected()
            return True
        if vk == VK_R:
            self._rotate_selected()
            return True
        if vk == VK_L:
            self.act_show_ratsnest.setChecked(not self.act_show_ratsnest.isChecked())
            self._on_toggle_ratsnest()
            return True
        if vk == VK_G:
            self.act_show_grid.setChecked(not self.act_show_grid.isChecked())
            self._toggle_grid()
            return True
        if vk == VK_F11:
            self.toggle_fullscreen()
            return True
        if vk == VK_F1:
            self._change_grid_step(-1)
            return True
        if vk == VK_F2:
            self._change_grid_step(1)
            return True
        if vk == VK_F3:
            self._toggle_grid_series()
            return True
        if vk in PAN_KEYS:
            self.canvas.pressed_keys.add(vk)
            if not self.canvas.pan_timer.isActive():
                self.canvas.pan_timer.start()
            return True
        return False

    def _handle_key_release(self, event):
        vk = event.nativeVirtualKey()
        if vk in PAN_KEYS:
            self.canvas.pressed_keys.discard(vk)
            if not self.canvas.pressed_keys:
                self.canvas.pan_timer.stop()
            return True
        return False

    def _load_grid_mode(self):
        mode = self._settings.value(SETTINGS_KEY_BRD_GRID_MODE, "1x", type=str)
        return mode if mode in GRID_SERIES else "1x"

    def _persist_grid_mode(self, mode):
        self._settings.setValue(SETTINGS_KEY_BRD_GRID_MODE, mode)
        self._settings.sync()

    def _apply_grid_series(self, mode, initial=False):
        if mode not in GRID_SERIES:
            return
        self.grid_mode = mode
        self._persist_grid_mode(mode)
        for m, act in self._grid_actions.items():
            act.setChecked(m == mode)
        idx = self.grid_series_combo.findData(mode)
        if idx >= 0 and self.grid_series_combo.currentIndex() != idx:
            self.grid_series_combo.blockSignals(True)
            self.grid_series_combo.setCurrentIndex(idx)
            self.grid_series_combo.blockSignals(False)
        values = GRID_SERIES[mode]
        self.grid_combo.blockSignals(True)
        self.grid_combo.clear()
        self.grid_combo.addItems(values)
        default = "1.000" if mode == "1x" else "1.270"
        self.grid_combo.setCurrentText(default if default in values else values[0])
        self.grid_combo.blockSignals(False)
        self.canvas.set_grid_size(float(self.grid_combo.currentText()))
        if not initial:
            self.show_log(f"Ряд сетки: {GRID_SERIES_LABEL[mode]}")

    def _toggle_grid_series(self):
        self._apply_grid_series("1.27x" if self.grid_mode == "1x" else "1x")

    def _load_library_paths(self):

    def _refresh_library_modules(self):
        for path in self.library_paths:
            if path not in self.library_modules:
                try:
                    footprints = parse_kicad_mod(path)
                    self.library_modules[path] = {fp.name: fp for fp in footprints}
                except Exception as e:
                    self.show_log(f"Ошибка библиотеки {Path(path).name}: {e}")

    def _find_footprint(self, name):
        for lib_data in self.library_modules.values():
            if name in lib_data:
                return lib_data[name]
        return None

    def _on_import_net(self):
        path, _ = QFileDialog.getOpenFileName(self, "Открыть сетлист", "",
                                              "Netlist (*.net);;All Files (*)")
        if not path:
            return
        self._refresh_library_modules()
        try:
            components = parse_netlist(path)
        except Exception as e:
            QMessageBox.critical(self, "Ошибка", f"Не удалось разобрать сетлист: {e}")
            return

        self.manager.clear()
        self.board = Board(name=Path(path).stem)
        self.manager.board = self.board

        missing = []
        next_uid = 1
        for comp in components:
            fp = self._find_footprint(comp.footprint)
            if fp is None:
                missing.append(f"{comp.reference} ({comp.footprint})")
                continue
            module = BoardModule(
                footprint=fp, reference=comp.reference, value=comp.value,
                x=0.0, y=0.0, angle=0.0, layer="F.Cu", uid=next_uid,
            )
            next_uid += 1
            pad_numbers = {p.number for p in fp.pads}
            for pad_number, net_name in comp.pins.items():
                if pad_number not in pad_numbers or not net_name:
                    continue
                net = self.board.get_or_create_net(net_name)
                module.pad_nets[pad_number] = PadNet(
                    pad_number=pad_number,
                    net_code=net.code,
                    net_name=net.name,
                )
            self.board.modules.append(module)
            self.manager.add_module(module)
        self._refresh_module_list()
        self.manager.rebuild_ratsnest()
        self.setWindowTitle(f"TraceCAD: {self.board.name}")
        self.show_log(f"Импортировано модулей: {len(self.board.modules)}")
        self._reset_history()
        if missing:
            QMessageBox.information(self, "Импорт завершён",
                "Импорт выполнен, но не найдены посадочные места:\n" + "\n".join(missing))

    def _on_open_board(self):

                                              "KiCad Board (*.brd);;All Files (*)")

            return
        self._load_board_from_path(path)

    def _load_board_from_path(self, path):
        try:
            board = parse_brd(path)
        except Exception as e:
            QMessageBox.critical(self, "Ошибка", f"Не удалось открыть плату: {e}")
            return
        self.manager.clear()
        self.board = board
        self.manager.board = self.board

        for module in self.board.modules:
            self.manager.add_module(module)
        for line in self.board.lines:
            self.manager.add_line(line)
        for circle in self.board.circles:
            self.manager.add_circle(circle)
        for arc in self.board.arcs:
            self.manager.add_arc(arc)
        for text in self.board.texts:
            self.manager.add_text(text)

        self._refresh_module_list()
        self.manager.rebuild_ratsnest()
        self.board.filename = path
        self.setWindowTitle(f"TraceCAD: {Path(path).name}")
        self.show_log(f"Плата загружена: {Path(path).name} ({len(self.board.modules)} модулей)")
        self._reset_history()

    def _on_save_board(self):
        path, _ = QFileDialog.getSaveFileName(self, "Сохранить плату", "",
                                              "KiCad Board (*.brd);;All Files (*)")
        if not path:
            return
        if not path.lower().endswith('.brd'):
            path += '.brd'
        try:
            text = build_brd_text(self.board)
            Path(path).write_text(text, encoding='utf-8')
            self.board.filename = path
            self.show_log(f"Плата сохранена: {Path(path).name}")
        except Exception as e:
            QMessageBox.critical(self, "Ошибка", f"Не удалось сохранить: {e}")

        self.current_tool = tool
        self._line_start = None
        self._net_start = None
        self.canvas.clear_ghost_line()

        if tool == BoardTool.SELECT:
            self._set_selectable_all(True)
            self.aux_bar.set_select_active(True)
            self.tool_bar.clear_active()
            self.canvas.setDragMode(self.canvas.DragMode.RubberBandDrag)
            self.right_panel.properties_panel.clear_panel()
        else:
            self._set_selectable_all(False)
            self.canvas.scene.clearSelection()
            self.aux_bar.set_select_active(False)
            self.tool_bar.set_active(tool)
            self.canvas.setDragMode(self.canvas.DragMode.NoDrag)
            if tool == BoardTool.ADD_LINE:
                self.right_panel.properties_panel.show_line_layer_picker(self._line_layer)
            elif tool == BoardTool.ADD_MODULE:
                self.right_panel.properties_panel.show_module_picker(self.library_modules)
            elif tool == BoardTool.ADD_NET:
                self.right_panel.properties_panel.clear_panel()
        self.show_log(f"Инструмент: {tool}")

    def _on_add_line_click(self, x, y):
        if self._line_start is None:
            self._line_start = (x, y)
            self.canvas.set_ghost_line(x, y, x, y)
            return
        sx, sy = self._line_start
        if abs(sx - x) < 1e-6 and abs(sy - y) < 1e-6:
            return
        line = BoardLine(
            x1=sx, y1=sy, x2=x, y2=y,
            layer=self._line_layer, width=DEFAULT_LINE_WIDTH,
            uid=self._next_line_uid(),
        )
        self.board.lines.append(line)
        self.manager.add_line(line)
        if self.current_tool != BoardTool.SELECT:
            try:
                line.graphics_item.setFlag(QGraphicsItem.GraphicsItemFlag.ItemIsSelectable, False)
                line.graphics_item.setFlag(QGraphicsItem.GraphicsItemFlag.ItemIsMovable, False)
            except RuntimeError:
                pass
        self._line_start = (x, y)
        self.canvas.set_ghost_line(x, y, x, y)
        self._commit_snapshot()
        self.show_log(f"Линия {self._line_layer}: добавлена")

    def _on_add_module_click(self, x, y):
        fp = self._selected_footprint
        if fp is None:
            self.show_log("Сначала выберите модуль справа")
            return
        import copy
        module = BoardModule(
            footprint=copy.deepcopy(fp),
            reference=self.board.auto_reference("REF"),
            value=fp.value,
            x=x, y=y, angle=0.0, layer="F.Cu",
            uid=self._next_module_uid(),
        )
        self.board.modules.append(module)
        self.manager.add_module(module)
        if self.current_tool != BoardTool.SELECT:
            try:
                module.graphics_item.setFlag(QGraphicsItem.GraphicsItemFlag.ItemIsSelectable, False)
                module.graphics_item.setFlag(QGraphicsItem.GraphicsItemFlag.ItemIsMovable, False)
            except RuntimeError:
                pass
        self.manager.rebuild_ratsnest()
        self._refresh_module_list()
        self._commit_snapshot()
        self.show_log(f"Модуль {module.reference} добавлен")

    def _on_add_net_click(self, x, y):
        tol = self.canvas.grid_size * 1.5
        hit = self.board.find_pad_at(x, y, tol)
        if hit is None:
            self.show_log("Не найден вывод в точке")
            return
        module, pad_number = hit
        pos = module.get_pad_absolute(pad_number)
        if self._net_start is None:
            self._net_start = (module, pad_number, pos)
            self.canvas.set_ghost_line(pos[0], pos[1], pos[0], pos[1])
            return
        m1, p1, pos1 = self._net_start
        if (m1, p1) == (module, pad_number):
            self._net_start = None
            self.canvas.clear_ghost_line()
            return
        pn1 = m1.pad_nets.get(p1)
        pn2 = module.pad_nets.get(pad_number)
        net_name = None
        if pn1 and pn1.net_name:
            net_name = pn1.net_name
        elif pn2 and pn2.net_name:
            net_name = pn2.net_name
        if net_name is None:
            net_name = f"Net-{m1.reference}-{p1}-{module.reference}-{pad_number}"
        net = self.board.get_or_create_net(net_name)
        for m, p in ((m1, p1), (module, pad_number)):
            m.pad_nets[p] = PadNet(pad_number=p, net_code=net.code, net_name=net.name)
        self._net_start = None
        self.canvas.clear_ghost_line()
        self.manager.rebuild_ratsnest()
        self._commit_snapshot()
        self.show_log(f"Связь {net_name}: {m1.reference}.{p1} ↔ {module.reference}.{pad_number}")

    def _delete_selected(self):
        deleted_modules = []
        deleted_lines = []
        deleted_circles = []
        deleted_arcs = []
        deleted_texts = []
        deleted_ratsnest = []
        for item in list(self.canvas.scene.selectedItems()):
            if isinstance(item, BoardModuleGraphicsItem):
                deleted_modules.append(item.module)
            elif isinstance(item, BoardLineGraphicsItem):
                deleted_lines.append(item.model)
            elif isinstance(item, BoardCircleGraphicsItem):
                deleted_circles.append(item.model)
            elif isinstance(item, BoardArcGraphicsItem):
                deleted_arcs.append(item.model)
            elif isinstance(item, BoardTextGraphicsItem):
                deleted_texts.append(item.model)
            elif isinstance(item, RatsnestGraphicsItem):
                deleted_ratsnest.append(item)

        deleted_net_codes = set()
        for r in deleted_ratsnest:
            if r.net_code > 0:
                deleted_net_codes.add(r.net_code)

        for m in deleted_modules:
            self.manager.remove_module(m)
            if m in self.board.modules:
                self.board.modules.remove(m)
        for l in deleted_lines:
            self.manager.remove_line(l)
            if l in self.board.lines:
                self.board.lines.remove(l)
        for c in deleted_circles:
            self.manager.remove_circle(c)
            if c in self.board.circles:
                self.board.circles.remove(c)
        for a in deleted_arcs:
            self.manager.remove_arc(a)
            if a in self.board.arcs:
                self.board.arcs.remove(a)
        for t in deleted_texts:
            self.manager.remove_text(t)
            if t in self.board.texts:
                self.board.texts.remove(t)
        for r in deleted_ratsnest:
            if r in self.manager.ratsnest_items:
                self.manager.ratsnest_items.remove(r)
            try:
                self.canvas.scene.removeItem(r)
            except RuntimeError:
                pass
        if deleted_net_codes:
            for module in self.board.modules:
                for pad_number in list(module.pad_nets.keys()):
                    if module.pad_nets[pad_number].net_code in deleted_net_codes:
                        del module.pad_nets[pad_number]

        if deleted_modules or deleted_lines or deleted_circles or deleted_arcs or deleted_texts or deleted_ratsnest:
            self.manager.rebuild_ratsnest()
            self._refresh_module_list()
            self._commit_snapshot()
            self.show_log(
                f"Удалено: {len(deleted_modules)} модулей, "
                f"{len(deleted_lines)} линий, {len(deleted_circles)} окружностей, "
                f"{len(deleted_arcs)} дуг, {len(deleted_texts)} текстов, "
                f"{len(deleted_ratsnest)} связей"
            )

    def _rotate_selected(self):
        items = [it for it in self.canvas.scene.selectedItems()
                 if isinstance(it, BoardModuleGraphicsItem)]
        if not items:
            return
        for item in items:
            item.module.angle = (item.module.angle + 90.0) % 360.0
            item.setRotation(item.module.angle)
            item.prepareGeometryChange()
            item.update()
        self.manager.rebuild_ratsnest()
        self._commit_snapshot()

    def _snapshot(self):
        import copy
        return {
            "modules": copy.deepcopy(self.board.modules),
            "lines": copy.deepcopy(self.board.lines),
            "circles": copy.deepcopy(self.board.circles),
            "arcs": copy.deepcopy(self.board.arcs),
            "texts": copy.deepcopy(self.board.texts),
            "nets": copy.deepcopy(self.board.nets),
        }

    def _restore_snapshot(self, snap):
        import copy
        self.board.modules = copy.deepcopy(snap["modules"])
        self.board.lines = copy.deepcopy(snap["lines"])
        self.board.circles = copy.deepcopy(snap["circles"])
        self.board.arcs = copy.deepcopy(snap["arcs"])
        self.board.texts = copy.deepcopy(snap.get("texts", []))
        self.board.nets = copy.deepcopy(snap["nets"])
        self.manager.clear()
        self.manager.board = self.board
        for m in self.board.modules:
            self.manager.add_module(m)
        for l in self.board.lines:
            self.manager.add_line(l)
        for c in self.board.circles:
            self.manager.add_circle(c)
        for a in self.board.arcs:
            self.manager.add_arc(a)
        for t in self.board.texts:
            self.manager.add_text(t)
        if self.current_tool != BoardTool.SELECT:
            self._set_selectable_all(False)
        self.manager.rebuild_ratsnest()
        self._refresh_module_list()

    def _reset_history(self):
        self._undo_stack = []
        self._redo_stack = []
        self._current_snapshot = self._snapshot()
        self._update_history_buttons()

    def _commit_snapshot(self):
        if self._current_snapshot is None:
            self._current_snapshot = self._snapshot()
            return
        self._undo_stack.append(self._current_snapshot)
        if len(self._undo_stack) > 100:
            self._undo_stack.pop(0)
        self._redo_stack.clear()
        self._current_snapshot = self._snapshot()
        self._update_history_buttons()

    def undo(self):
        if not self._undo_stack:
            return
        self._redo_stack.append(self._current_snapshot)
        self._current_snapshot = self._undo_stack.pop()
        self._restore_snapshot(self._current_snapshot)
        self._update_history_buttons()
        self.show_log("отменено")

    def redo(self):
        if not self._redo_stack:
            return
        self._undo_stack.append(self._current_snapshot)
        self._current_snapshot = self._redo_stack.pop()
        self._restore_snapshot(self._current_snapshot)
        self._update_history_buttons()
        self.show_log("повторено")

    def _update_history_buttons(self):
        self.aux_bar.set_undo_enabled(bool(self._undo_stack))
        self.aux_bar.set_redo_enabled(bool(self._redo_stack))

    def _restore_last_board(self):
        last = self._settings.value(SETTINGS_KEY_BRD_LAST, "", type=str)
        if not last:
            self._reset_history()
            return
        p = Path(last)
        if not p.exists() or not p.is_file():
            self._reset_history()
            return
        self._load_board_from_path(str(p))

        if self.board.filename:
            self._settings.setValue(SETTINGS_KEY_BRD_LAST, self.board.filename)
        else:
            self._settings.setValue(SETTINGS_KEY_BRD_LAST, "")
        self._settings.sync()

    def closeEvent(self, event):
        self._save_last_board()

                self._build_arc_ui(obj.model)
                self.pins_label.hide()
                self.pins_table.hide()
                self.buttons_widget.show()
            elif isinstance(obj, BoardTextGraphicsItem):
                self.header_label.setText(f"Текст {obj.model.layer}")
                self._build_text_ui(obj.model)
                self.pins_label.hide()
                self.pins_table.hide()
                self.buttons_widget.show()

                self.header_label.setText(f"Связь #{obj.net_code}")
                self._build_ratsnest_ui(obj)
                self.pins_label.hide()
                self.pins_table.hide()
                self.buttons_widget.hide()

                self.clear_panel()

            self.form_widget.show()
            self.line_layers_widget.hide()
            self.module_picker_widget.hide()

            self.updateGeometry()
