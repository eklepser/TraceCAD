import re
from pathlib import Path
from footprint_editor.core.models import (
    Footprint, Pad, PadShape, PadType, Line, Circle, Text, TextType
)

KICAD_UNIT = 1000  # 1 unit = 1/1000 mm (mils)

def kicad_to_mm(value):
    return float(value) / KICAD_UNIT

def parse_kicad_mod(filepath):
    content = Path(filepath).read_text(encoding='utf-8', errors='ignore')
    return parse_kicad_content(content)

def parse_kicad_content(content):
    footprints = []
    modules = re.findall(r'\$MODULE\s+(\S+)(.*?)\$EndMODULE', content, re.DOTALL)
    
    for name, body in modules:
        fp = parse_module(name, body)
        footprints.append(fp)
    
    return footprints

def parse_module(name, body):
    fp = Footprint(name=name)
    
    t0_match = re.search(r'T0\s+(.*?)\n', body)
    t1_match = re.search(r'T1\s+(.*?)\n', body)
    
    if t0_match:
        ref_text = parse_text_line(t0_match.group(1), TextType.REFERENCE)
        ref_text.locked = True
        fp.texts.append(ref_text)
        fp.reference = ref_text.text
    
    if t1_match:
        val_text = parse_text_line(t1_match.group(1), TextType.VALUE)
        val_text.locked = True
        fp.texts.append(val_text)
        fp.value = val_text.text
    
    for ds_match in re.finditer(r'DS\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)\s+(\d+)', body):
        x1, y1, x2, y2, thickness = ds_match.groups()
        line = Line(
            x1=kicad_to_mm(x1), y1=kicad_to_mm(y1),
            x2=kicad_to_mm(x2), y2=kicad_to_mm(y2),
            width=kicad_to_mm(thickness)
        )
        fp.lines.append(line)
    
    for dc_match in re.finditer(r'DC\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)\s+(\d+)', body):
        x, y, r, _, thickness = dc_match.groups()
        circle = Circle(
            x=kicad_to_mm(x), y=kicad_to_mm(y),
            radius=kicad_to_mm(r),
            width=kicad_to_mm(thickness)
        )
        fp.circles.append(circle)
    
    for da_match in re.finditer(r'DA\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)\s+(-?\d+)\s+(\d+)\s+(\d+)', body):
        x1, y1, x2, y2, angle, thickness = da_match.groups()
        line = Line(
            x1=kicad_to_mm(x1), y1=kicad_to_mm(y1),
            x2=kicad_to_mm(x2), y2=kicad_to_mm(y2),
            width=kicad_to_mm(thickness)
        )
        fp.lines.append(line)
    
    pads = re.findall(r'\$PAD(.*?)\$EndPAD', body, re.DOTALL)
    for pad_body in pads:
        pad = parse_pad(pad_body)
        if pad:
            fp.pads.append(pad)
    
    return fp

def parse_text_line(text_str, text_type):
    text_match = re.search(r'"([^"]*)"', text_str)
    text = text_match.group(1) if text_match else ""
    
    params_str = text_str[:text_match.start()].strip() if text_match else text_str
    params = params_str.split()
    
    x = kicad_to_mm(params[0]) if len(params) > 0 else 0
    y = kicad_to_mm(params[1]) if len(params) > 1 else 0
    w = kicad_to_mm(params[2]) if len(params) > 2 else 1
    h = kicad_to_mm(params[3]) if len(params) > 3 else 1
    size = max(w, h)
    
    visible = True
    if len(params) > 7:
        visible = params[7] != 'Y'
    
    return Text(
        x=x, y=y, text=text, size=size,
        visible=visible, text_type=text_type, locked=True
    )

def parse_pad(pad_body):
    sh_match = re.search(r'Sh\s+"(\d+)"\s+(\w+)\s+(\d+)\s+(\d+)', pad_body)
    dr_match = re.search(r'Dr\s+(\d+)', pad_body)
    po_match = re.search(r'Po\s+(-?\d+)\s+(-?\d+)', pad_body)
    at_match = re.search(r'At\s+(\w+)', pad_body)
    
    if not sh_match or not po_match:
        return None
    
    number = sh_match.group(1)
    shape_str = sh_match.group(2)
    size_x = kicad_to_mm(sh_match.group(3))
    size_y = kicad_to_mm(sh_match.group(4))
    x = kicad_to_mm(po_match.group(1))
    y = kicad_to_mm(po_match.group(2))
    drill = kicad_to_mm(dr_match.group(1)) if dr_match else 0
    
    shape_map = {'R': PadShape.RECT, 'C': PadShape.CIRCLE, 'O': PadShape.OVAL}
    shape = shape_map.get(shape_str, PadShape.CIRCLE)
    
    pad_type = PadType.SMD
    if at_match:
        attr = at_match.group(1)
        if attr in ('STD', 'NPTH'):
            pad_type = PadType.THT
    
    return Pad(
        number=number, shape=shape, pad_type=pad_type,
        x=x, y=y, width=size_x, height=size_y,
        drill=drill if drill > 0 else None
    ).

from PySide6.QtWidgets import QGraphicsItem
from PySide6.QtCore import Qt

class BaseGraphicsItem(QGraphicsItem):
    def __init__(self, model, canvas, parent=None):
        super().__init__(parent)
        self.model = model
        self.canvas = canvas
        self.model.graphics_item = self
        
        self.setFlag(QGraphicsItem.GraphicsItemFlag.ItemIsSelectable, True)
        
        self._drag_start_mouse = None
        self._is_drag_leader = False
        self._group_drag_start = {}

    def _collect_selected_items(self):
        self._group_drag_start = {}
        for item in self.scene().selectedItems():
            cls_name = type(item).__name__
            if cls_name == 'PadGraphicsItem':
                self._group_drag_start[item] = ('pad', item.model.x, item.model.y)
            elif cls_name == 'LineGraphicsItem':
                self._group_drag_start[item] = ('line', item.model.x1, item.model.y1, item.model.x2, item.model.y2)
            elif cls_name == 'CircleGraphicsItem':
                self._group_drag_start[item] = ('circle', item.model.x, item.model.y)
            elif cls_name == 'TextGraphicsItem':
                self._group_drag_start[item] = ('text', item.model.x, item.model.y)

    def mousePressEvent(self, event):
        if event.button() == Qt.MouseButton.LeftButton and self.isSelected():
            self._drag_start_mouse = self.mapToScene(event.pos())
            self._is_drag_leader = True
            self._collect_selected_items()
        super().mousePressEvent(event)

    def mouseMoveEvent(self, event):
        if self._is_drag_leader and (event.buttons() & Qt.MouseButton.LeftButton):
            current_mouse = self.mapToScene(event.pos())
            delta = current_mouse - self._drag_start_mouse
            grid = self.canvas.grid_size
            snapped_dx = round(delta.x() / grid) * grid
            snapped_dy = round(delta.y() / grid) * grid
            
            for item, start in self._group_drag_start.items():
                item.apply_drag_delta(start, snapped_dx, snapped_dy)
        else:
            super().mouseMoveEvent(event)

    def mouseReleaseEvent(self, event):
        self._is_drag_leader = False
        self._group_drag_start = {}
        super().mouseReleaseEvent(event)

    def update_geometry(self):
        self.prepareGeometryChange()
        self.update()

    def apply_drag_delta(self, start, dx, dy):
        raise NotImplementedError.

class PadGraphicsItem(BaseGraphicsItem):
    def __init__(self, pad, canvas, parent=None):
        super().__init__(pad, canvas, parent)
        self.setPos(pad.x, pad.y)

    def boundingRect(self):
        w = self.model.width
        h = self.model.height
        extra = 0.5
        return QRectF(-w/2-extra, -h/2-extra, w+2*extra, h+2*extra)

    def paint(self, painter, option, widget):
        if option.state & QStyle.StateFlag.State_Selected:
            color = QColor(*Colors.SELECTED)
        else:
            color = QColor(*Colors.PAD)
        
        w = self.model.width
        h = self.model.height
        drill = self.model.drill or 0.0
        
        painter.setPen(Qt.PenStyle.NoPen)
        painter.setBrush(color)
        
        if self.model.shape == PadShape.CIRCLE:
            painter.drawEllipse(QRectF(-w/2, -h/2, w, h))
        elif self.model.shape == PadShape.RECT:
            painter.drawRect(QRectF(-w/2, -h/2, w, h))
        elif self.model.shape == PadShape.OVAL:
            painter.drawRoundedRect(QRectF(-w/2, -h/2, w, h), w/2, h/2)
        
        if drill > 0:
            painter.setBrush(QColor(*Colors.BACKGROUND))
            painter.drawEllipse(QRectF(-drill/2, -drill/2, drill, drill))

    def drawBackground(self, painter, rect):
        super().drawBackground(painter, rect)
        scale_factor = self.transform().m11()
        cell_px = self.grid_size * scale_factor
        
        if cell_px < Sizes.MIN_CELL_PX:
            return

        left = int(rect.left() // self.grid_size) * self.grid_size
        right = int(rect.right() // self.grid_size) * self.grid_size
        top = int(rect.top() // self.grid_size) * self.grid_size
        bottom = int(rect.bottom() // self.grid_size) * self.grid_size
        
        draw_step = self.grid_size
        max_points = Sizes.MAX_GRID_POINTS
        
        if (right - left) / draw_step > max_points:
            factor = 2
            while (right - left) / (draw_step * factor) > max_points: factor *= 2
            draw_step = self.grid_size * factor
        if (bottom - top) / draw_step > max_points:
            factor = 2
            while (bottom - top) / (draw_step * factor) > max_points: factor *= 2
            draw_step = self.grid_size * factor

        pen = QPen(self.grid_color, 0)
        painter.setPen(pen)
        
        x = left
        while x <= right:
            y = top
            while y <= bottom:
                painter.drawPoint(QPointF(x, y))
                y += draw_step
            x += draw_step

    def drawForeground(self, painter, rect):
        super().drawForeground(painter, rect)
        scale = self.transform().m11()
        cross_size = Sizes.CROSS_SIZE / scale

        axis_pen = QPen(QColor(*Colors.AXIS), 0)
        axis_pen.setCosmetic(True)
        painter.setPen(axis_pen)
        painter.drawLine(QPointF(rect.left(), 0), QPointF(rect.right(), 0))
        painter.drawLine(QPointF(0, rect.top()), QPointF(0, rect.bottom()))

        if self._snap_pos and self.show_snap:
            pen = QPen(QColor(*Colors.SNAP_CROSS), 0)
            pen.setCosmetic(True)
            painter.setPen(pen)
            x, y = self._snap_pos.x(), self._snap_pos.y()
            painter.drawLine(QPointF(x - cross_size, y), QPointF(x + cross_size, y))
            painter.drawLine(QPointF(x, y - cross_size), QPointF(x, y + cross_size))

        if self._ghost_line_start and self._ghost_line_end:
            pen = QPen(QColor(*Colors.GHOST_LINE), 0)
            pen.setCosmetic(True)
            pen.setStyle(Qt.PenStyle.DashLine)
            painter.setPen(pen)
            painter.drawLine(self._ghost_line_start, self._ghost_line_end)

        if self._ghost_circle_center and self._ghost_circle_radius > 0:
            pen = QPen(QColor(*Colors.GHOST_LINE), 0)
            pen.setCosmetic(True)
            pen.setStyle(Qt.PenStyle.DashLine)
            painter.setPen(pen)
            painter.setBrush(Qt.BrushStyle.NoBrush)
            painter.drawEllipse(self._ghost_circle_center, self._ghost_circle_radius, self._ghost_circle_radius)

    def wheelEvent(self, event: QWheelEvent):
        if event.angleDelta().y() > 0:
            if self.zoom_index < len(self.zoom_steps) - 1: self.zoom_index += 1
        else:
            if self.zoom_index > 0: self.zoom_index -= 1

class GlobalKeyFilter(QObject):
    def __init__(self, window):
        super().__init__()
        self.window = window
        
    def eventFilter(self, obj, event):
        if event.type() in (QEvent.Type.KeyPress, QEvent.Type.KeyRelease):
            focus_widget = QApplication.focusWidget()
            if focus_widget and isinstance(focus_widget, (QLineEdit, QDoubleSpinBox)):
                return super().eventFilter(obj, event)
            return self.window.handle_global_key(event)
        return super().eventFilter(obj, event)

    def _export_to_kicad(self, filepath):
        from datetime import datetime
        
        lines = []
        lines.append(f"PCBNEW-LibModule-V1  {datetime.now().strftime('%d.%m.%Y %H:%M:%S')}")
        lines.append("# encoding utf-8")
        lines.append("$INDEX")
        lines.append(self.footprint.name)
        lines.append("$EndINDEX")
        lines.append(f"$MODULE {self.footprint.name}")
        lines.append("Po 0 0 0 15 00000000 00000000 ~~")
        lines.append(f"Li {self.footprint.name}")
        lines.append("Sc 00000000")
        lines.append("AR")
        lines.append("Op 0 0 0")
        
        for text in self.footprint.texts:
            if text.text_type == TextType.REFERENCE:
                idx = 0
            elif text.text_type == TextType.VALUE:
                idx = 1
            else:
                continue
            
            x_k = int(text.x * 1000)
            y_k = int(text.y * 1000)
            size_k = int(text.size * 1000)
            hide = 'Y' if not text.visible else 'N'
            lines.append(f'T{idx} {x_k} {y_k} {size_k} {size_k} 0 120 N N N 21 N {hide} "{text.text}"')
        
        for line in self.footprint.lines:
            x1 = int(line.x1 * 1000)
            y1 = int(line.y1 * 1000)
            x2 = int(line.x2 * 1000)
            y2 = int(line.y2 * 1000)
            t = int(line.width * 1000)
            lines.append(f"DS {x1} {y1} {x2} {y2} {t} 21")
        
        for circle in self.footprint.circles:
            x = int(circle.x * 1000)
            y = int(circle.y * 1000)
            r = int(circle.radius * 1000)
            t = int(circle.width * 1000)
            lines.append(f"DC {x} {y} {r} 0 {t} 21")
        
        for pad in self.footprint.pads:
            lines.append("$PAD")
            shape_map = {PadShape.RECT: 'R', PadShape.CIRCLE: 'C', PadShape.OVAL: 'O'}
            shape_char = shape_map.get(pad.shape, 'C')
            sx = int(pad.width * 1000)
            sy = int(pad.height * 1000)
            drill = int((pad.drill or 0) * 1000)
            x = int(pad.x * 1000)
            y = int(pad.y * 1000)
            attr = 'STD' if pad.pad_type == PadType.THT else 'SMD'
            lines.append(f'Sh "{pad.number}" {shape_char} {sx} {sy} 0 0 0')
            lines.append(f"Dr {drill} 0 0")
            lines.append(f"At {attr} N 00E0FFFF")
            lines.append('Ne 0 ""')
            lines.append(f"Po {x} {y}")
            lines.append("$EndPAD")
        
        lines.append(f"$EndMODULE  {self.footprint.name}")
        lines.append("$EndLIBRARY")
        
        Path(filepath).write_text('\n'.join(lines), encoding='utf-8')

    def _on_canvas_clicked(self, x, y):
        if self.current_tool == Tool.ADD_PAD:
            next_num = self.manager.get_next_pad_number()
            new_pad = Pad(number=next_num, shape=PadShape.CIRCLE, pad_type=PadType.SMD, x=x, y=y, width=1.5, height=1.5, drill=0.8)
            self.footprint.pads.append(new_pad)
            self.manager.add_model(new_pad)
            self.canvas.set_interaction_enabled(False)
            self.show_log(f'элемент "площадка" #{next_num} установлен')
            
        elif self.current_tool == Tool.ADD_LINE:
            if self.line_start_point is None:
                self.line_start_point = (x, y)
                self.canvas.set_ghost_line(x, y, x, y)
            else:
                start_x, start_y = self.line_start_point
                new_line = Line(start_x, start_y, x, y)
                self.footprint.lines.append(new_line)
                self.manager.add_model(new_line)
                self.canvas.set_interaction_enabled(False)
                self.line_start_point = (x, y)
                self.canvas.set_ghost_line(x, y, x, y)
                self.show_log('элемент "линия" установлен')

        elif self.current_tool == Tool.ADD_CIRCLE:
            if self.circle_center_point is None:
                self.circle_center_point = (x, y)
                self.canvas.set_ghost_circle(x, y, 0.0)
            else:
                cx, cy = self.circle_center_point
                r = math.sqrt((x - cx)**2 + (y - cy)**2)
                r = round(r / self.canvas.grid_size) * self.canvas.grid_size
                if r > 0:
                    new_circle = Circle(cx, cy, r)
                    self.footprint.circles.append(new_circle)
                    self.manager.add_model(new_circle)
                    self.canvas.set_interaction_enabled(False)
                    self.show_log('элемент "окружность" установлен')
                self.circle_center_point = None
                self.canvas.clear_ghost_circle()

        elif self.current_tool == Tool.ADD_TEXT:
            self.text_position = (x, y)
            self.canvas.set_ghost_text(x, y, "", 1.0, True)
            self._show_text_dialog()

    def _show_text_dialog(self):
        dialog = TextDialog(self)
        dialog.text_changed.connect(lambda text: self.canvas.set_ghost_text(*self.text_position, text, dialog.get_size(), dialog.get_visible()))
        dialog.size_changed.connect(lambda size: self.canvas.set_ghost_text(*self.text_position, dialog.get_text(), size, dialog.get_visible()))
        dialog.visible_changed.connect(lambda visible: self.canvas.set_ghost_text(*self.text_position, dialog.get_text(), dialog.get_size(), visible))
        
        if dialog.exec() == QDialog.Accepted:
            text = dialog.get_text()
            size = dialog.get_size()
            visible = dialog.get_visible()
            if text:
                new_text = Text(self.text_position[0], self.text_position[1], text, size, visible)
                self.footprint.texts.append(new_text)
                self.manager.add_model(new_text)
                self.canvas.set_interaction_enabled(False)
                self.show_log('элемент "текст" установлен')
        
        self.text_position = None
        self.canvas.clear_ghost_text()

    def _on_mouse_moved(self, x, y):
        self._current_mouse_scene_pos = QPointF(x, y)
        self.coords_label.setText(f"X: {x:.4f}, Y: {y:.4f}")
        
        if self.current_tool == Tool.ADD_LINE and self.line_start_point:
            self.canvas.set_ghost_line(self.line_start_point[0], self.line_start_point[1], x, y)
        elif self.current_tool == Tool.ADD_CIRCLE and self.circle_center_point:
            cx, cy = self.circle_center_point
            r = math.sqrt((x - cx)**2 + (y - cy)**2)
            r = round(r / self.canvas.grid_size) * self.canvas.grid_size
            self.canvas.set_ghost_circle(cx, cy, r)
        else:
            self.canvas.clear_ghost_line()
            self.canvas.clear_ghost_circle()

    def _on_selection_changed(self):
        selected_items = self.canvas.scene.selectedItems()
        has_selection = len(selected_items) > 0
        self.right_panel.btn_delete.setEnabled(has_selection)
        
        if len(selected_items) == 1:
            item = selected_items[0]
            if isinstance(item, (PadGraphicsItem, LineGraphicsItem, CircleGraphicsItem, TextGraphicsItem)):
                if getattr(item.model, 'locked', False):
                    self.right_panel.btn_delete.setEnabled(False)
                self.right_panel.properties_panel.set_object(item.model)
            else:
                self.right_panel.properties_panel.clear_panel()
        else:
            self.right_panel.properties_panel.clear_panel()

    def _delete_selected(self):
        selected_items = self.canvas.scene.selectedItems()
        for item in selected_items:
            if isinstance(item, (PadGraphicsItem, LineGraphicsItem, CircleGraphicsItem, TextGraphicsItem)):
                if getattr(item.model, 'locked', False):
                    continue
                if item.model in self.footprint.pads:
                    self.footprint.pads.remove(item.model)
                if item.model in self.footprint.lines:
                    self.footprint.lines.remove(item.model)
                if item.model in self.footprint.circles:
                    self.footprint.circles.remove(item.model)
                if item.model in self.footprint.texts:
                    self.footprint.texts.remove(item.model)
                self.manager.remove_model(item.model)
                
        self.right_panel.btn_delete.setEnabled(False)
        self.right_panel.properties_panel.clear_panel()
        self.show_log("элементы удалены")

    def _copy_selected(self):
        self.clipboard = []
        for item in self.canvas.scene.selectedItems():
            if isinstance(item, (PadGraphicsItem, LineGraphicsItem, CircleGraphicsItem, TextGraphicsItem)):
                if not getattr(item.model, 'locked', False):
                    self.clipboard.append(copy.deepcopy(item.model))
        if self.clipboard:
            self.show_log("(буфер обмена) элементы скопированы")

    def _paste_from_clipboard(self):
        if not self.clipboard:
            return
        self.canvas.scene.clearSelection()
        
        center_x, center_y = 0, 0
        count = 0
        for model in self.clipboard:
            if isinstance(model, Pad):
                center_x += model.x; center_y += model.y; count += 1
            elif isinstance(model, Line):
                center_x += (model.x1 + model.x2)/2; center_y += (model.y1 + model.y2)/2; count += 1
            elif isinstance(model, Circle):
                center_x += model.x; center_y += model.y; count += 1
            elif isinstance(model, Text):
                center_x += model.x; center_y += model.y; count += 1
                
        if count > 0:
            center_x /= count
            center_y /= count
            
        dx = self._current_mouse_scene_pos.x() - center_x
        dy = self._current_mouse_scene_pos.y() - center_y
        dx = round(dx / self.canvas.grid_size) * self.canvas.grid_size
        dy = round(dy / self.canvas.grid_size) * self.canvas.grid_size
        
        for model in self.clipboard:
            new_model = copy.deepcopy(model)
            new_model.uid = str(uuid.uuid4())
            new_model.locked = False
            
            if isinstance(new_model, Pad):
                new_model.x += dx; new_model.y += dy
                self.footprint.pads.append(new_model)
            elif isinstance(new_model, Line):
                new_model.x1 += dx; new_model.y1 += dy
                new_model.x2 += dx; new_model.y2 += dy
                self.footprint.lines.append(new_model)
            elif isinstance(new_model, Circle):
                new_model.x += dx; new_model.y += dy
                self.footprint.circles.append(new_model)
            elif isinstance(new_model, Text):
                new_model.x += dx; new_model.y += dy
                self.footprint.texts.append(new_model)
            
            item = self.manager.add_model(new_model)
            item.setSelected(True)
                
        self.show_log("(буфер обмена) элементы установлены")

    def handle_global_key(self, event):
        vk = event.nativeVirtualKey()
        modifiers = event.modifiers()
        
        if event.type() == QEvent.Type.KeyPress:
            if vk in PAN_KEYS:
                self.canvas.pressed_keys.add(vk)
                if not self.canvas.pan_timer.isActive():
                    self.canvas.pan_timer.start()
                return True
            elif vk == VK_ESCAPE:
                self._finish_drawing()
                self.canvas.scene.clearSelection()
                self.right_panel.btn_select.setChecked(True)
                self._on_tool_button_clicked(self.right_panel.btn_select)
                return True
            elif vk == VK_DELETE:
                self._delete_selected()
                return True
            elif vk == VK_F1:
                self._change_grid_step(-1)
                return True
            elif vk == VK_F2:
                self._change_grid_step(1)
                return True
            elif vk == VK_C and (modifiers & Qt.KeyboardModifier.ControlModifier):
                self._copy_selected()
                return True
            elif vk == VK_V and (modifiers & Qt.KeyboardModifier.ControlModifier):
                self._paste_from_clipboard()
                return True
                
        elif event.type() == QEvent.Type.KeyRelease:
            if vk in PAN_KEYS:
                self.canvas.pressed_keys.discard(vk)
                if not self.canvas.pressed_keys:
                    self.canvas.pan_timer.stop()
                return True
                
        return False.
